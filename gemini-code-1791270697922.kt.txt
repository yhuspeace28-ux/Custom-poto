package com.fotorapi.app.ai

import android.content.Context
import android.graphics.Bitmap
import ai.onnxruntime.OnnxTensor
import ai.onnxruntime.OrtEnvironment
import ai.onnxruntime.OrtSession
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.nio.FloatBuffer

class InpainterEngine(private val context: Context) {

    private var ortEnv: OrtEnvironment? = null
    private var ortSession: OrtSession? = null

    fun initModel(modelFileName: String = "lama_inpainting_fp16.onnx") {
        ortEnv = OrtEnvironment.getEnvironment()
        val modelBytes = context.assets.open(modelFileName).readBytes()

        val options = OrtSession.SessionOptions().apply {
            addNnapi()
        }
        ortSession = ortEnv?.createSession(modelBytes, options)
    }

    suspend fun removeObject(imageBitmap: Bitmap, maskBitmap: Bitmap): Bitmap = withContext(Dispatchers.Default) {
        val width = 512
        val height = 512

        val scaledImage = Bitmap.createScaledBitmap(imageBitmap, width, height, true)
        val scaledMask = Bitmap.createScaledBitmap(maskBitmap, width, height, true)

        val imageBuffer = bitmapToFloatBuffer(scaledImage, isMask = false)
        val maskBuffer = bitmapToFloatBuffer(scaledMask, isMask = true)

        val imageShape = longArrayOf(1, 3, height.toLong(), width.toLong())
        val maskShape = longArrayOf(1, 1, height.toLong(), width.toLong())

        val imageTensor = OnnxTensor.createTensor(ortEnv, imageBuffer, imageShape)
        val maskTensor = OnnxTensor.createTensor(ortEnv, maskBuffer, maskShape)

        val inputs = mapOf("image" to imageTensor, "mask" to maskTensor)
        val results = ortSession?.run(inputs)

        val outputTensor = results?.get(0)?.value as Array<Array<FloatArray>>
        val resultBitmap = tensorToBitmap(outputTensor, width, height)

        imageTensor.close()
        maskTensor.close()
        results.close()

        return@withContext resultBitmap
    }

    private fun bitmapToFloatBuffer(bitmap: Bitmap, isMask: Boolean): FloatBuffer {
        val width = bitmap.width
        val height = bitmap.height
        val pixels = IntArray(width * height)
        bitmap.getPixels(pixels, 0, width, 0, 0, width, height)

        val channels = if (isMask) 1 else 3
        val buffer = FloatBuffer.allocate(1 * channels * height * width)

        if (isMask) {
            for (pixel in pixels) {
                val r = (pixel shr 16) and 0xFF
                buffer.put(if (r > 128) 1.0f else 0.0f)
            }
        } else {
            for (pixel in pixels) buffer.put(((pixel shr 16) and 0xFF) / 255.0f)
            for (pixel in pixels) buffer.put(((pixel shr 8) and 0xFF) / 255.0f)
            for (pixel in pixels) buffer.put((pixel and 0xFF) / 255.0f)
        }
        buffer.rewind()
        return buffer
    }

    private fun tensorToBitmap(tensor: Array<Array<FloatArray>>, width: Int, height: Int): Bitmap {
        val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)
        val pixels = IntArray(width * height)

        var idx = 0
        for (y in 0 until height) {
            for (x in 0 until width) {
                val r = (tensor[0][y][x].coerceIn(0.0f, 1.0f) * 255).toInt()
                val g = (tensor[1][y][x].coerceIn(0.0f, 1.0f) * 255).toInt()
                val b = (tensor[2][y][x].coerceIn(0.0f, 1.0f) * 255).toInt()

                pixels[idx++] = (0xFF shl 24) or (r shl 16) or (g shl 8) or b
            }
        }
        bitmap.setPixels(pixels, 0, width, 0, 0, width, height)
        return bitmap
    }

    fun close() {
        ortSession?.close()
        ortEnv?.close()
    }
}
